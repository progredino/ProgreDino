        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';

        let app, auth, db;

        if (firebaseConfig) {

            try {

                app = firebase.initializeApp(firebaseConfig);

                auth = firebase.auth();

                db = firebase.firestore();

            } catch (e) {

                console.error("Firebase init error", e);

        // --- CONSTANTS & INITIAL STATE ---

        const CYCLE_DAYS = 7;

        const INITIAL_TASKS = [

            { id: 'gym', name: 'Atividade física', icon: 'dumbbell', requiredPerWeek: 3, completions: 0, completedToday: false, color: 'bg-blue-500' },

            { id: 'study', name: 'Momento de Estudo', icon: 'book-open', requiredPerWeek: 5, completions: 0, completedToday: false, color: 'bg-purple-500' },

            { id: 'news', name: 'Giro de Notícias', icon: 'newspaper', requiredPerWeek: 2, completions: 0, completedToday: false, color: 'bg-yellow-500' },

            { id: 'meals', name: 'Refeições (Café, Almoço, Jantar)', icon: 'utensils', requiredPerWeek: 7, completions: 0, completedToday: false, color: 'bg-orange-500' },

        ];

        const MAX_POSSIBLE_ACTIONS = INITIAL_TASKS.length * 7;

        let state = {

            tasks: JSON.parse(JSON.stringify(INITIAL_TASKS)),

            daysElapsed: 0,

            level: 1,

            xp: 0,

            isGameOver: false,

            showConfetti: false,

            popupMessage: null,

            activeTab: 'tracker',

            activeSkin: 'base',

            totalStats: { gym: 0, study: 0, news: 0, meals: 0 },

            isFirebaseLoading: !!firebaseConfig,

            userId: null,

            authStatus: 'checking',

            loginError: null

        // --- STATE MANAGEMENT ---

        function updateState(newState) {

            state = { ...state, ...newState };

            render();

            saveStateToFirebase();

        }

        // --- FIREBASE LOGIC ---

        let isInitialLoadDone = false;

        if (auth) {

            const token = typeof __initial_auth_token !== 'undefined' ? __initial_auth_token : null;

            if (token && !auth.currentUser) {

                auth.signInWithCustomToken(token).catch(e => console.error("Erro token:", e));

            }

            auth.onAuthStateChanged(async (user) => {

                if (user) {

                    state.userId = user.uid;

                    state.authStatus = 'authenticated';

                    try {

                        const docRef = db.collection('artifacts').doc(appId).collection('users').doc(user.uid).collection('gameState').doc('current');

                        const docSnap = await docRef.get();

                       

                        if (docSnap.exists) {

                            const data = docSnap.data();

                            if (data.tasks) {

                                state.tasks = INITIAL_TASKS.map(initialTask => {

                                    const savedTask = data.tasks.find(t => t.id === initialTask.id);

                                    return savedTask ? { ...initialTask, completions: savedTask.completions, completedToday: savedTask.completedToday } : { ...initialTask };

                                });

                            }

                            if (data.daysElapsed !== undefined) state.daysElapsed = data.daysElapsed;

                            if (data.level !== undefined) state.level = data.level;

                            if (data.xp !== undefined) state.xp = data.xp;

                            else if (data.happiness !== undefined) state.xp = data.happiness;

                           

                            if (data.isGameOver !== undefined) state.isGameOver = data.isGameOver;

                            if (data.totalStats) state.totalStats = data.totalStats;

                            if (data.activeSkin) state.activeSkin = data.activeSkin;

                        }

                    } catch (e) {

                        console.error("Erro ao carregar dados salvos", e);

                    isInitialLoadDone = true;

                    state.isFirebaseLoading = false;

                    render();

                } else {

                    state.userId = null;

                    state.authStatus = 'login';

                    render();

                }

            state.isFirebaseLoading = false;

        async function saveStateToFirebase() {

            if (state.isFirebaseLoading || !state.userId || !db || !isInitialLoadDone) return;

            try {

                const tasksToSave = state.tasks.map(({ icon, ...rest }) => rest);

                const docRef = db.collection('artifacts').doc(appId).collection('users').doc(state.userId).collection('gameState').doc('current');

               

                await docRef.set({

                    tasks: tasksToSave,

                    daysElapsed: state.daysElapsed,

                    xp: state.xp,

                    level: state.level,

                    isGameOver: state.isGameOver,

                    totalStats: state.totalStats,

                    activeSkin: state.activeSkin

                });

            } catch (e) {

                console.error("Erro ao salvar progresso", e);

            }

        }

        function getHappiness() {

            const totalCompleted = state.tasks.reduce((sum, task) => sum + task.completions, 0);

            return Math.min(100, Math.round((totalCompleted / MAX_POSSIBLE_ACTIONS) * 100));

        }

        function getDinoState(happiness) {

            if (happiness < 20) return 'sad';

            if (happiness < 50) return 'neutral';

            if (happiness < 80) return 'happy';

            return 'super-happy';

        }

        function handleTaskComplete(taskId) {

            if (state.isGameOver) return;

            const taskIndex = state.tasks.findIndex(t => t.id === taskId);

            const task = state.tasks[taskIndex];

           

            if (!task || task.completions >= 7 || task.completedToday) return;

            const newTasks = [...state.tasks];

            newTasks[taskIndex] = { ...task, completions: task.completions + 1, completedToday: true };

           

            let newXp = state.xp + (100 / MAX_POSSIBLE_ACTIONS);

            let newLevel = state.level;

            let showConfetti = state.showConfetti;

            if (newXp >= 100) {

                newLevel = Math.min(state.level + 1, 50);

                newXp = newXp - 100;

                showConfetti = true;

                setTimeout(() => {

                    updateState({ showConfetti: false });

                }, 3000);

            }

            const newTotalStats = { ...state.totalStats, [taskId]: state.totalStats[taskId] + 1 };

            updateState({ tasks: newTasks, xp: newXp, level: newLevel, showConfetti, totalStats: newTotalStats });

        }

        function handleTaskRemove(taskId) {

            if (state.isGameOver) return;

            const taskIndex = state.tasks.findIndex(t => t.id === taskId);

            const task = state.tasks[taskIndex];

           

            if (!task || task.completions === 0) return;

            const newTasks = [...state.tasks];

            newTasks[taskIndex] = { ...task, completions: task.completions - 1, completedToday: false };

            const newXp = Math.max(0, state.xp - (100 / MAX_POSSIBLE_ACTIONS));

            const newTotalStats = { ...state.totalStats, [taskId]: Math.max(0, state.totalStats[taskId] - 1) };

            updateState({ tasks: newTasks, xp: newXp, totalStats: newTotalStats });

        }

        function advanceDay() {

            const happiness = getHappiness();

           

            if (state.daysElapsed >= CYCLE_DAYS - 1) {

                let msg = "Ainda não me acostumei com a fama😎";

                if (happiness <= 50) msg = "Acho que você confundiu Fim da semana com Fim do Mundo☄️";

                else if (happiness <= 70) msg = "Não sou perfeito...mas tô quase lá🫡";

               

                updateState({ isGameOver: true, popupMessage: msg });

            } else {

                const nextDay = state.daysElapsed + 1;

                const newTasks = state.tasks.map(task => ({ ...task, completedToday: false }));

               

                let msg = null;

                if (nextDay === 2) {

                    if (happiness <= 10) msg = "Que semaninha, ein?😢";

                    else if (happiness <= 15) msg = "Acho que tô pegando o ritmo🤔";

                    else msg = "Eu sou o 'Milior'🤖";

                } else if (nextDay === 4) {

                    if (happiness <= 30) msg = "Oi...Eu ainda tô aqui, tá?💀";

                    else if (happiness <= 44) msg = "O jogo só acaba quando termina🥵";

                    else msg = "Eu devia ser estudado pela NASA🛸";

                }

                updateState({ daysElapsed: nextDay, tasks: newTasks, popupMessage: msg });

            }

        }

        }
